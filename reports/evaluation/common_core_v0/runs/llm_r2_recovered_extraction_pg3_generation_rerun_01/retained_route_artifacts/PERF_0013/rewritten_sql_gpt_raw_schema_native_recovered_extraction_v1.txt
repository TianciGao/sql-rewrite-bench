select 	n_name, 	sum(l_extendedprice * (1 - l_discount)) as revenue from 	customer, 	orders, 	lineitem, 	supplier, 	nation, 	region where 	c_custkey = o_custkey 	and l_orderkey = o_orderkey 	and l_suppkey = s_suppkey 	and c_nationkey = s_nationkey 	and s_nationkey = n_nationkey 	and n_regionkey = r_regionkey 	and r_name = 'MIDDLE EAST' 	and o_orderdate >= date '1997-01-01' 	and o_orderdate < date '1997-01-01' + interval '1' year group by 	n_name order by 	revenue desc SELECT nation.n_name, SUM(lineitem.l_extendedprice * (1 - lineitem.l_discount)) AS revenue FROM customer, orders, lineitem, supplier, nation, region WHERE customer.c_custkey = orders.o_custkey AND lineitem.l_orderkey = orders.o_orderkey AND (lineitem.l_suppkey = supplier.s_suppkey AND customer.c_nationkey = supplier.s_nationkey) AND (supplier.s_nationkey = nation.n_nationkey AND nation.n_regionkey = region.r_regionkey AND (region.r_name = 'MIDDLE EAST' AND (orders.o_orderdate >= DATE '1997-01-01' AND orders.o_orderdate < (DATE '1997-01-01' + INTERVAL '1' YEAR)))) GROUP BY nation.n_name ORDER BY SUM(lineitem.l_extendedprice * (1 - lineitem.l_discount)) DESC
